Overview

What this system does

  • Fetches real-time weather observations from KMA API Hub (UltraSrtNcst).
  • Parses JSON categories: T1H(Temp), REH(Humidity), RN1(Rain), VEC(Wind direction), WSD(Wind speed).
  • Displays values on the OLED (SSD1306).
  • Turns on LEDs based on weather conditions (Sunny/Windy/Rainy).

My role in the group

  • OLED Text UI: formatted 4-line weather output.
  • LED Control Logic: rule-based mapping for Rainy / Windy / Sunny.
  • Debugging: verified JSON parsing and handled base_time logic (NTP sync).
  • Project Page: documented implementation details & demo for evaluation.

Implementation Details (My Part)

A) OLED Weather Output

  • Libraries: Adafruit_SSD1306, Adafruit_GFX, Wire.
  • Display layout: 4 lines (Temp / Humid / Rain / Wind).
  • Wind format: VEC(deg) + WSD(m/s) for clarity.
  • Update policy: periodic fetch (e.g., every 60 seconds) or on demand.
SSD1306 Wire SDA=21 SCL=22 printf layout

B) LED Indicator Logic

  • LED mapping:
    • Blue = Rainy
    • Yellow = Windy
    • Red = Sunny
  • Priority rule: Rainy first → then Windy → then Sunny (prevents conflicts).
  • Why priority? Rain is the most critical condition for immediate action (umbrella alert).
Rule-based Priority Readable conditions

C) How I ensured the code is correct (defense points)

  • Time sync: used NTP (KST offset) and adjusted base_time when minutes < 15.
  • Data validation: extracted only known categories (T1H/REH/RN1/VEC/WSD).
  • Serial logs: printed URL + parsed values for verification and debugging.
  • OLED check: verified formatting fits 128×64 (text size 1).

const int   daylightOffset_sec = 0;

// =======================================
// setup()
// =======================================
void setup() {
  Serial.begin(115200);
  delay(500);

  // OLED 초기화
  Wire.begin(21, 22);  // ESP32: SDA=21, SCL=22
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED INIT FAIL!");
    while (1);
  }
  display.clearDisplay();
  display.display();

  // WiFi
  Serial.print("Connecting to WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(300);
  }
  Serial.println("\nWiFi Connected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());

  // NTP
  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  Serial.println("Waiting NTP...");
  struct tm timeinfo;
  while (!getLocalTime(&timeinfo)) {
    Serial.println("Retry...");
    delay(500);
  }
  Serial.println("Time synced!");

  // 첫 조회
  getWeatherNow();
}

void loop() {
  // 필요 시 주기 호출
  delay(60000);
  getWeatherNow();
}


// ========================================================
// 기상청 API 요청 생성
// ========================================================
void getWeatherNow() {
  if (WiFi.status() != WL_CONNECTED) return;

  struct tm timeinfo;
  getLocalTime(&timeinfo);

  time_t now = mktime(&timeinfo);
  if (timeinfo.tm_min < 15) now -= 3600;
  struct tm* baseTm = localtime(&now);

  char baseDate[9];
  sprintf(baseDate, "%04d%02d%02d",
          baseTm->tm_year + 1900,
          baseTm->tm_mon + 1,
          baseTm->tm_mday);
  String base_date = baseDate;

  char baseTime[5];
  sprintf(baseTime, "%02d00", baseTm->tm_hour);
  String base_time = baseTime;

  // URL 구성
  String url = String(host) +
    "/api/typ02/openApi/VilageFcstInfoService_2.0/getUltraSrtNcst"
    "?authKey=" + authKey +
    "&dataType=JSON"
    "&numOfRows=60"
    "&pageNo=1"
    "&base_date=" + base_date +
    "&base_time=" + base_time +
    "&nx=" + nx +
    "&ny=" + ny;

  Serial.println("Request URL:");
  Serial.println(url);

  WiFiClientSecure client;
  client.setInsecure();

  HTTPClient http;
  http.setTimeout(20000);

  if (!http.begin(client, url)) {
    Serial.println("HTTP begin fail");
    return;
  }

  int code = http.GET();
  if (code == HTTP_CODE_OK) {
    parseWeatherJson(http.getString(), base_date, base_time);
  } else {
    Serial.print("HTTP ERROR: ");
    Serial.println(code);
  }
  http.end();
}


// ========================================================
// JSON 파싱 + OLED 출력
// ========================================================
void parseWeatherJson(const String& json, const String& base_date, const String& base_time) {
  DynamicJsonDocument doc(16384);
  DeserializationError error = deserializeJson(doc, json);

  if (error) {
    Serial.println("JSON error");
    return;
  }

  JsonVariant items = doc["response"]["body"]["items"]["item"];
  if (items.isNull()) {
    Serial.println("No data");
    return;
  }

  float T1H = NAN; 
  float REH = NAN;
  float RN1 = NAN;
  float VEC = NAN;
  float WSD = NAN;

  for (JsonObject item : items.as<JsonArray>()) {
    String cat = item["category"];
    float  val = String(item["obsrValue"]).toFloat();

    if (cat == "T1H") T1H = val;
    if (cat == "REH") REH = val;
    if (cat == "RN1") RN1 = val;
    if (cat == "VEC") VEC = val;
    if (cat == "WSD") WSD = val;
  }

  // OLED 출력
  display.clearDisplay();
  display.setTextSize(1); 
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);

  display.printf("Temp: %.1f C\n", T1H);  
  display.printf("Humid: %.0f %%\n", REH);  
  display.printf("Rain: %.1f mm\n", RN1);  
  display.printf("Wind: %.0f%c / %.1f m/s\n", VEC, 248, WSD); // 248 = °
  display.display();
}

// Loop → Serial 입력 체크
// =============================
void loop() {
  if (Serial.available()) {
    int newX = Serial.parseInt();
    int newY = Serial.parseInt();

    if (newX > 0 && newY > 0) {
      nx = newX;
      ny = newY;
      getWeatherNow();
    }
  }
  delay(100);
}

// =============================
// JSON 파싱 + OLED + LED
// =============================
void parseWeatherJson(const String& json) {
  DynamicJsonDocument doc(16384);

  if (deserializeJson(doc, json)) return;

  JsonVariant items = doc["response"]["body"]["items"]["item"];
  if (items.isNull()) return;

  float T1H=NAN, REH=NAN, RN1=NAN, WSD=NAN, VEC=NAN;

  for (JsonObject it : items.as<JsonArray>()) {
    String cat = it["category"];
    float v = String(it["obsrValue"]).toFloat();

    if (cat=="T1H") T1H=v;
    if (cat=="REH") REH=v;
    if (cat=="RN1") RN1=v;
    if (cat=="VEC") VEC=v;
    if (cat=="WSD") WSD=v;
  }

  // LED 점등 (우선순위: Rainy > Windy > Sunny)
  if (RN1 > 0) {
    digitalWrite(LED_BLUE, HIGH);
    digitalWrite(LED_RED, LOW);
    digitalWrite(LED_YELLOW, LOW);
  }
  else if (VEC >= 3.5) {
    digitalWrite(LED_YELLOW, HIGH);
    digitalWrite(LED_RED, LOW);
    digitalWrite(LED_BLUE, LOW);
  }
  else if (RN1 == 0 && REH <= 50) {
    digitalWrite(LED_RED, HIGH);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_BLUE, LOW);
  }
  else {
    digitalWrite(LED_RED, LOW);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_BLUE, LOW);
  }

  // OLED 출력
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);

  display.printf("Temp: %.1f C\n", T1H);
  display.printf("Humid: %.0f %%\n", REH);
  display.printf("Rain: %.1f mm\n", RN1);
  display.printf("Wind: %.0fdeg / %.1fm/s\n", VEC, WSD);

  display.display();
}

Defense / FAQ (prof-style questions)

Why use NTP and base_time adjustment?
KMA ultra-short observations are updated at fixed times. If the current minute is < 15, the latest available dataset may still be the previous hour, so I adjust the base time for reliable fetching.
How do you know parsing is correct?
I print the request URL and parsed values (T1H/REH/RN1/VEC/WSD) to Serial, then verify they match expected units and OLED output format.
Why prioritize Rainy over Windy/Sunny?
Rain is a high-impact condition. Priority prevents conflicting LED states and improves user clarity: if rain exists, show Rainy regardless of other values.

What I will defend in the final evaluation

  • Correctness: proven by serial logs + OLED values + LED states.
  • Clarity: readable rules, consistent units, and clean OLED layout.
  • Integration: KMA API + WiFi + NTP + OLED + LED together.
  • Maintainability: modular functions (getWeatherNow(), parseWeatherJson()).
