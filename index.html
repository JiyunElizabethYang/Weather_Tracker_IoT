Method (My Implementation)

A) OLED Weather Output

The OLED weather output was designed to present real-time weather information in a clear and compact format suitable for a 128×64 display. By selecting only essential weather variables and arranging them in a fixed four-line layout, the display focuses on clearly presenting numerical data, enabling quick and reliable inspection of current weather conditions.

  • Goal: Display the latest weather values on an ESP32 OLED.
  • UI layout: Temperature, Humid, Rain, Wind
  • Wind format: VECdeg / WSDm/s (human-readable and compact).

B) LED Indicator Rules

The LED thresholds were established by reviewing standard meteorological references, such as the Beaufort wind force scale and official weather guidelines, and by considering how these criteria are commonly interpreted in Korea. The final cutoffs were selected to reflect weather conditions that people in Korea typically experience in daily life, ensuring that the rules are both explainable and practical.

  • Rainy (Blue): RN1 > 0 (raining)
  • Windy (Yellow): WSD >= 3.5 (noticeably windy for daily life)
  • Sunny (Red): RN1 == 0 AND REH <= 50 (dry & bright condition)
Explainable rules Weather-based thresholds Compared to typical weather in Korea

How I Chose the LED Thresholds

  • Windy (Yellow): Based on the Beaufort wind force scale, a wind speed around 3.4–5.4 m/s corresponds to a gentle breeze, where leaves and small twigs are continuously moving. Since winds above 3.5 m/s are typically noticeable to people in daily life, I set the LED rule to turn Yellow LED ON when wind speed WSD ≤ 3.5 m/s.
  • Sunny (Red): In Korea, the commonly referenced comfortable/typical indoor humidity level is around 50%, and according to KMA-related criteria, a humidity range of 0–50% can be interpreted as a “clear/dry” condition. Therefore, I defined the “Sunny” state as no rainfall and humidity ≤ 50%, turning the Red LED ON when RN1 == 0 and REH ≤ 50.

void fetchAndParseWeather() {
  if (WiFi.status() != WL_CONNECTED) return;

  // base date/time (if minutes < 15, use previous hour)
  struct tm timeinfo;
  getLocalTime(&timeinfo);
  time_t now = mktime(&timeinfo);
  if (timeinfo.tm_min < 15) now -= 3600;
  struct tm* bt = localtime(&now);

  char baseDate[9];
  sprintf(baseDate, "%04d%02d%02d", bt->tm_year + 1900, bt->tm_mon + 1, bt->tm_mday);

  char baseTime[5];
  sprintf(baseTime, "%02d00", bt->tm_hour);

  String url = String(host) +
    "/api/typ02/openApi/VilageFcstInfoService_2.0/getUltraSrtNcst"
    "?authKey=" + String(authKey) +
    "&dataType=JSON"
    "&numOfRows=60"
    "&pageNo=1"
    "&base_date=" + String(baseDate) +
    "&base_time=" + String(baseTime) +
    "&nx=" + nx +
    "&ny=" + ny;

  Serial.println("\nRequest URL:");
  Serial.println(url);

  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient http;
  http.setTimeout(20000);

  if (!http.begin(client, url)) return;

  int code = http.GET();
  if (code == HTTP_CODE_OK) {
    String json = http.getString();

    DynamicJsonDocument doc(16384);
    if (deserializeJson(doc, json)) {
      Serial.println("JSON parse error");
      http.end();
      return;
    }

    JsonVariant items = doc["response"]["body"]["items"]["item"];
    if (items.isNull()) {
      Serial.println("No items");
      http.end();
      return;
    }

    // reset
    T1H = REH = RN1 = VEC = WSD = NAN;

    for (JsonObject it : items.as<JsonArray>()) {
      String cat = it["category"];
      float v = String(it["obsrValue"]).toFloat();

      if (cat == "T1H") T1H = v;
      if (cat == "REH") REH = v;
      if (cat == "RN1") RN1 = v;
      if (cat == "VEC") VEC = v;
      if (cat == "WSD") WSD = v;
    }

    Serial.println("Parsed values updated!");
  } else {
    Serial.print("HTTP ERROR: ");
    Serial.println(code);
  }

  http.end();
}

(B) LED Indicator Code

// LEDs
#define LED_RED     14   // Sunny
#define LED_YELLOW  12   // Windy
#define LED_BLUE    27   // Rainy

void setupLEDs() {
  pinMode(LED_RED, OUTPUT);
  pinMode(LED_YELLOW, OUTPUT);
  pinMode(LED_BLUE, OUTPUT);
}

// Uses the latest parsed values: RN1, REH, WSD
void updateLEDs() {
  // Rainy (Blue): precipitation detected
  if (RN1 > 0) {
    digitalWrite(LED_BLUE, HIGH);
    digitalWrite(LED_YELLOW, LOW);
    digitalWrite(LED_RED, LOW);
    Serial.println("LED: Rainy (Blue)");
    return;
  }

  // Windy (Yellow): windy condition
  if (WSD >= 3.5) {
    digitalWrite(LED_YELLOW, HIGH);
    digitalWrite(LED_BLUE, LOW);
    digitalWrite(LED_RED, LOW);
    Serial.println("LED: Windy (Yellow)");
    return;
  }

  // Sunny (Red): dry & bright condition
  if (RN1 == 0 && REH <= 50) {
    digitalWrite(LED_RED, HIGH);
    digitalWrite(LED_BLUE, LOW);
    digitalWrite(LED_YELLOW, LOW);
    Serial.println("LED: Sunny (Red)");
    return;
  }

  // Otherwise: OFF
  digitalWrite(LED_RED, LOW);
  digitalWrite(LED_YELLOW, LOW);
  digitalWrite(LED_BLUE, LOW);
  Serial.println("LED: OFF");
}
